    // components/ops/api-types.ts  (types only; if lib/ops/** from 03.2-04 exports any of these, re-export it instead)
    export type Locale3 = "en" | "ar" | "es";
    export type TranslationState = "published" | "draft" | "missing";
    export type TextStatus = "published" | "draft";
    export type MediaRef = { id: string; key: string; url: string; width: number | null; height: number | null; alt_en: string };
    export type Translations<T> = Record<Locale3, (T & { status: TextStatus }) | null>;
    export type DestinationSummary / DestinationDetail / StaySummary / StayDetail / StayText / RatesView / Block /
      StayAccess / CatalogSummary / CatalogDetail / TeamSummary / TeamDetail / JourneySummary / JourneyDetail
      = exactly contract §5.1-5.8, field for field.
    export type PublishResult = { id: string; is_published: boolean; affects_site: true; warnings: { code: "no_base_rate" | "no_price" }[] };

    // components/ops/ops-client.ts
    export type OpsError = { code: string; field: string | null; locale: Locale3 | null; detail: unknown };
    export type OpsResponse<T> = ({ ok: true } & T) | { ok: false; status: number; error: OpsError };
    export function opsFetch<T>(path: `/api/ops/${string}`, query?: Record<string, string | undefined>): Promise<OpsResponse<T>>;
    export function opsSend<T>(path: `/api/ops/${string}`, body: unknown): Promise<OpsResponse<T>>;   // POST JSON
    //  - credentials "same-origin", header content-type application/json, cache "no-store"
    //  - 403 with code not_owner -> window.location.assign("/sign-in") and the promise still resolves (ok false)
    //  - network failure -> { ok: false, status: 0, error: { code: "network", field: null, locale: null, detail: null } }
    //  - a success body that carries `site` -> window.dispatchEvent(new CustomEvent("almar:site", { detail: body.site }))
    //    (03.2-05's status line may listen; the kit does not depend on it)

    // components/ops/use-dashboard-locale.ts
    export function useDashboardLocale(): Locale3;   // "en" on the server and first paint, then the cookie value

    export type MissingItem = { locale: Locale3 | null; field: string };   // field names as in contract §5.9
    export function missingForPublish(entity: PublishEntity, draft: {
      translations: Partial<Record<Locale3, Record<string, unknown> | null>>;
      media_id: string | null;                  // hero / photo / image, per entity
      destination_ids?: string[];               // catalog_item
      destination_published?: boolean;         // stay
    }): MissingItem[];

      dirty: boolean;                    // Close with unsaved changes asks "Discard changes?" (ConfirmDialog)

    }): JSX.Element;
