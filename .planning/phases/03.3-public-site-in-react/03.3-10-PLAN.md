```ts
// lib/data/types.ts — CatalogItem gains, right after stay_ids:
destination_slugs: string[];   // slugs of destination_ids, same order
stay_slugs: string[];          // slugs of stay_ids, same order

// lib/data/catalog-filter.ts (pure; imports: `import type` from "./types", `{ foldText }` from "./stay-filter")
export type CatalogFilter = { kind?: CatalogKind; destinations?: string[]; stays?: string[]; query?: string };
export type CatalogFilterable = Pick<CatalogItem, "kind" | "name" | "summary" | "destination_slugs" | "stay_slugs">;
export function matchesCatalogFilter(item: CatalogFilterable, f: CatalogFilter): boolean;
export function filterCatalog<T extends CatalogFilterable>(items: readonly T[], f: CatalogFilter): T[];
export type CatalogQuery = { kind?: CatalogKind; destinations: string[]; stays: string[]; item?: string };
export function parseCatalogQuery(
  params: URLSearchParams,
  opts?: { destinations?: readonly string[]; stays?: readonly string[]; items?: readonly string[] },
): CatalogQuery;   // keys: type (experience|service), destination (comma list), stay (comma list), item
export function toCatalogQuery(q: CatalogQuery): string;   // "" or "?type=…&destination=a,b&stay=x&item=…"

// lib/data/experiences.ts
getCatalogItems(locale, opts?: { kind?; destinationSlug?; destinationSlugs?: string[]; staySlug?; staySlugs?: string[]; query? })
```

Rule (design 3.2): kind exact when set; destinations any-of; stays any-of; all groups AND; an empty or absent
list is no constraint; query = `foldText(query.trim())`, matched as a substring of `foldText(name)` or
`foldText(summary ?? "")`, empty after trim = no constraint. Search text is never in the URL (CatalogQuery has
no query field). `parseCatalogQuery`: `type` only `experience` | `service`, else undefined; list values split
on `,`, trimmed, kept only when they match `^[a-z0-9]+(?:-[a-z0-9]+)*$`, de-duplicated in first-seen order,
at most 20 per list, and when `opts.destinations` / `opts.stays` is given, only members of it; `item` kept
only when slug-shaped (and in `opts.items` when given); a repeated key uses its first value (`params.get`);
`%2C` decodes to a comma through URLSearchParams, so both forms parse. `toCatalogQuery`: keys in the order
type, destination, stay, item; empty parts omitted; list values joined with a raw `,` (slugs need no
encoding); non-slug values dropped; returns `""` when nothing is set, else a string starting with `?`.
The optional `opts` argument is additive to the brief's signature (same pattern as `parseStayQuery`).
